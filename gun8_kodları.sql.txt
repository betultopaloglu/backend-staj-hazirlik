-- 1. YARIM KALAN TABLOLARI TEMİZLEME (Çakışmayı Önler)
DROP TABLE IF EXISTS work_logs CASCADE;
DROP TABLE IF EXISTS user_projects CASCADE;
DROP TABLE IF EXISTS projects CASCADE;
DROP TABLE IF EXISTS users CASCADE;

-- 2. TABLOLARI SIFIRDAN OLUŞTURMA
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL
);

CREATE TABLE projects (
    id SERIAL PRIMARY KEY,
    project_name VARCHAR(100) NOT NULL,
    owner_id INT REFERENCES users(id) ON DELETE SET NULL
);

CREATE TABLE user_projects (
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    project_id INT REFERENCES projects(id) ON DELETE CASCADE,
    PRIMARY KEY (user_id, project_id)
);

CREATE TABLE work_logs (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    project_id INT REFERENCES projects(id) ON DELETE CASCADE,
    work_date DATE NOT NULL,
    hours INT NOT NULL
);

/*Proje bazlı toplam ve en fazla çalışılan saatler*/
SELECT
    p.project_name,
    SUM(w.hours) AS toplam_calisma_saati,
    MAX(w.hours) AS en_fazla_tek_seferde_calisilan_saat
FROM work_logs w
JOIN projects p ON w.project_id = p.id
GROUP BY p.project_name;

/*Kullanıcıların toplam çalışma saatlerini büyükten küçüğe sırala*/
SELECT
    u.first_name,
    u.last_name,
    SUM(w.hours) AS genel_toplam_saat
FROM work_logs w
JOIN users u ON w.user_id = u.id
GROUP BY u.id, u.first_name, u.last_name
ORDER BY genel_toplam_saat DESC;
